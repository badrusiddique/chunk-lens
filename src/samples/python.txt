"""Retry decorator with exponential back-off and optional jitter."""

import time
import random
import functools
from typing import Callable, TypeVar, Any

F = TypeVar("F", bound=Callable[..., Any])


def retry(
    max_attempts: int = 3,
    base_delay: float = 0.5,
    max_delay: float = 30.0,
    jitter: bool = True,
    exceptions: tuple[type[Exception], ...] = (Exception,),
) -> Callable[[F], F]:
    """Retry a function with exponential back-off.

    Args:
        max_attempts: Total attempts including the first call.
        base_delay: Initial delay between retries in seconds.
        max_delay: Upper cap on delay regardless of back-off multiplier.
        jitter: If True, adds random noise to avoid thundering-herd issues.
        exceptions: Exception types that trigger a retry.

    Returns:
        A decorator that wraps the target callable.
    """

    def decorator(func: F) -> F:
        @functools.wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            last_exc: Exception | None = None
            for attempt in range(max_attempts):
                try:
                    return func(*args, **kwargs)
                except exceptions as exc:
                    last_exc = exc
                    if attempt == max_attempts - 1:
                        break
                    delay = min(base_delay * (2**attempt), max_delay)
                    if jitter:
                        delay *= 0.5 + random.random() * 0.5
                    time.sleep(delay)
            raise RuntimeError(
                f"{func.__name__} failed after {max_attempts} attempts"
            ) from last_exc

        return wrapper  # type: ignore[return-value]

    return decorator


# Example

@retry(max_attempts=5, base_delay=1.0, exceptions=(ConnectionError, TimeoutError))
def fetch_json(url: str) -> dict[str, Any]:
    import urllib.request, json
    with urllib.request.urlopen(url, timeout=10) as r:
        return json.loads(r.read().decode())
